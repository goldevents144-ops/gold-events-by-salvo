   201→function NotificationsCard() {
   202→  const [notifications, setNotifications] = React.useState<any[]>([]);
   203→  const [loading, setLoading] = React.useState(true);
   204→
   205→  const loadNotifications = React.useCallback(() => {
   206→    fetch("/api/admin/audit")
   207→      .then((r) => r.json())
   208→      .then((d) => {
   209→        const notifs = (d.logs || [])
   210→          .filter((l: any) =>
   211→            l.action === "notification_new_request" ||
   212→            l.action === "notification_new_partner"
   213→          )
   214→          .slice(0, 5);
   215→        setNotifications(notifs);
   216→      })
   217→      .finally(() => setLoading(false));
   218→  }, []);
   219→
   220→  React.useEffect(() => {
   221→    loadNotifications();
   222→    // Auto-refresh ogni 30 secondi per nuove notifiche
   223→    const interval = setInterval(loadNotifications, 30_000);
   224→    return () => clearInterval(interval);
   225→  }, [loadNotifications]);
   226→
   227→  return (
   228→    <div className="rounded-2xl border border-[#D4AF37]/30 bg-[#1C1C1C]/60 p-5 sm:p-6">
   229→      <div className="mb-4 flex items-center justify-between gap-3">
   230→        <div className="flex items-center gap-2">
   231→          <span className="relative inline-flex size-2.5">
   232→            <span className="absolute inline-flex size-full animate-ping rounded-full bg-[#D4AF37]/60" />
   233→            <span className="relative inline-flex size-2.5 rounded-full bg-[#D4AF37]" />
   234→          </span>
   235→          <h3 className="font-serif text-base font-semibold text-white sm:text-lg">
   236→            Notifiche recenti
   237→          </h3>
   238→        </div>
   239→        <span className="text-[10px] uppercase tracking-[0.2em] text-[#6B6B6B]">
   240→          Auto-refresh 30s
   241→        </span>
   242→      </div>
   243→
   244→      {loading ? (
   245→        <p className="py-4 text-center text-sm text-[#A1A1A1]">Caricamento…</p>
   246→      ) : notifications.length === 0 ? (
   247→        <div className="py-6 text-center">
   248→          <p className="text-sm text-[#A1A1A1]">Nessuna nuova richiesta.</p>
   249→          <p className="mt-1 text-[11px] text-[#6B6B6B]">
   250→            Quando un cliente compila il form, apparirà qui automaticamente.
   251→          </p>
   252→        </div>
   253→      ) : (
   254→        <ul className="space-y-2.5">
   255→          {notifications.map((n) => {
   256→            const meta = JSON.parse(n.meta || "{}");
   257→            const isPartner = n.action === "notification_new_partner";
   258→            return (
   259→              <li
   260→                key={n.id}
   261→                className="flex items-start gap-3 rounded-xl border border-white/8 bg-[#0F0F0F]/60 p-3"
   262→              >
   263→                <span
   264→                  className={`mt-0.5 inline-flex size-7 shrink-0 items-center justify-center rounded-lg ${
   265→                    isPartner
   266→                      ? "border border-blue-500/30 bg-blue-500/10 text-blue-400"
   267→                      : "border border-[#D4AF37]/30 bg-[#D4AF37]/10 text-[#D4AF37]"
   268→                  }`}
   269→                >
   270→                  {isPartner ? "🤝" : "📋"}
   271→                </span>
   272→                <div className="flex-1 text-sm">
   273→                  <p className="font-medium text-white">
   274→                    {isPartner
   275→                      ? meta.businessName || "Nuovo partner"
   276→                      : meta.fullName || "Nuova richiesta"}
   277→                  </p>
   278→                  <p className="text-xs text-[#A1A1A1]">
   279→                    {isPartner
   280→                      ? `${meta.activityType} · ${meta.location}`
   281→                      : `${meta.eventType} · ${meta.eventDate} · ${meta.guests} ospiti · ${meta.location}`}

   283→                  <p className="mt-0.5 text-[10px] uppercase tracking-[0.18em] text-[#6B6B6B]">
   284→                    {new Date(n.createdAt).toLocaleString("it-IT")}
   285→                  </p>
   286→                </div>

   393→function ContattiTab() {
   394→  const [items, setItems] = React.useState<any[]>([]);
   395→  const [loading, setLoading] = React.useState(true);
   396→  const load = () => {
   397→    setLoading(true);
   398→    fetch("/api/admin/submissions")
   399→      .then((r) => r.json())
   400→      .then((d) => setItems(d.submissions ?? []))
   401→      .finally(() => setLoading(false));
   402→  };
   403→  React.useEffect(load, []);
   404→
   405→  const updateStatus = async (id: string, status: string) => {
   406→    await fetch(`/api/admin/submissions/${id}`, {
   407→      method: "PATCH",
   408→      headers: { "Content-Type": "application/json" },
   409→      body: JSON.stringify({ status }),
   410→    });
   411→    load();
   412→  };
   413→
   414→  if (loading) {
   415→    return <Loader2 className="size-6 animate-spin text-[#D4AF37]" />;
   416→  }
   417→  if (items.length === 0) {
   418→    return (
   419→      <p className="rounded-xl border border-white/8 bg-[#1C1C1C]/40 p-8 text-center text-sm text-[#A1A1A1]">
   420→        Nessuna richiesta ancora. Le nuove richieste del form appariranno qui.
   421→      </p>
   422→    );
   423→  }
   424→  return (
   425→    <div className="flex flex-col gap-3">
   426→      {items.map((s) => (
   427→        <details
   428→          key={s.id}
   429→          className="group rounded-2xl border border-white/8 bg-[#1C1C1C]/60 p-4 transition-colors hover:border-[#D4AF37]/30"
   430→        >
   431→          <summary className="cursor-pointer list-none">
   432→            <div className="flex flex-wrap items-center justify-between gap-3">
   433→              <div>
   434→                <p className="font-semibold text-white">{s.fullName}</p>
   435→                <p className="text-xs text-[#A1A1A1]">
   436→                  {s.eventType} · {new Date(s.createdAt).toLocaleString("it-IT")}
   437→                </p>
   438→              </div>
   439→              <span className="rounded-full border border-[#D4AF37]/30 bg-[#D4AF37]/10 px-3 py-1 text-[10px] font-semibold uppercase tracking-[0.16em] text-[#D4AF37]">
   440→                {s.status}
   441→              </span>
   442→            </div>
   443→          </summary>
   444→          <div className="mt-4 grid gap-3 border-t border-white/8 pt-4 text-sm sm:grid-cols-2">
   445→            <Field label="Data evento" value={s.eventDate} />
   446→            <Field label="Location" value={s.location} />
   447→            <Field label="Ospiti" value={String(s.guests)} />
   448→            <Field label="Servizi" value={s.services} />
   449→            {s.notes && <Field label="Note" value={s.notes} />}
   450→            <div className="sm:col-span-2 flex flex-wrap gap-2 pt-2">
   451→              {["new", "read", "contacted", "archived"].map((st) => (
   452→                <button
   453→                  key={st}
   454→                  type="button"
   455→                  onClick={() => updateStatus(s.id, st)}
   456→                  className={`rounded-full px-3 py-1.5 text-[10px] font-semibold uppercase tracking-[0.14em] transition-colors ${
   457→                    s.status === st
   458→                      ? "bg-[#D4AF37] text-[#0F0F0F]"
   459→                      : "border border-white/12 text-[#A1A1A1] hover:border-[#D4AF37]/50 hover:text-white"
   460→                  }`}
   461→                >
   462→                  {st}
   463→                </button>
   464→              ))}
   465→            </div>
   466→          </div>
   467→        </details>
   468→      ))}

   522→function AiTab() {
   523→  const [input, setInput] = React.useState("");
   524→  const [messages, setMessages] = React.useState<
   525→    Array<{ role: "user" | "assistant"; content: string }>
   526→  >([]);
   527→  const [loading, setLoading] = React.useState(false);
   528→
   529→  const send = async () => {
   530→    const text = input.trim();
   531→    if (!text || loading) return;
   532→    setInput("");
   533→    setMessages((m) => [...m, { role: "user", content: text }]);
   534→    setLoading(true);
   535→    try {
   536→      const res = await fetch("/api/admin/ai", {
   537→        method: "POST",
   538→        headers: { "Content-Type": "application/json" },
   539→        body: JSON.stringify({ question: text }),
   540→      });
   541→      const data = await res.json();
   542→      setMessages((m) => [
   543→        ...m,
   544→        { role: "assistant", content: data.reply || "(errore)" },
   545→      ]);
   546→    } catch {
   547→      setMessages((m) => [
   548→        ...m,
   549→        { role: "assistant", content: "Errore di connessione." },
   550→      ]);
   551→    } finally {
   552→      setLoading(false);
   553→    }
   554→  };
   555→
   556→  return (
   557→    <div className="mx-auto flex max-w-2xl flex-col gap-4">
   558→      <div className="rounded-2xl border border-[#D4AF37]/20 bg-[#1C1C1C]/40 p-5">
   559→        <h2 className="font-serif text-lg font-semibold text-white">
   560→          Assistente AI
   561→        </h2>
   562→        <p className="mt-1 text-sm text-[#A1A1A1]">
   563→          Chiedimi come modificare il sito, quali prezzi aggiornare, come gestire i contenuti.
   564→          Ti darò suggerimenti — ogni modifica deve essere confermata manualmente.
   565→        </p>
   566→      </div>
   567→      <div className="flex max-h-[400px] min-h-[200px] flex-col gap-2 overflow-y-auto rounded-2xl border border-white/8 bg-[#1C1C1C]/60 p-4">
   568→        {messages.length === 0 && (
   569→          <p className="text-center text-sm text-[#A1A1A1]">
   570→            Es: "Come posso aggiornare il prezzo di Luxury Wedding?"
   571→          </p>
   572→        )}
   573→        {messages.map((m, i) => (
   574→          <div
   575→            key={i}
   576→            className={`flex ${m.role === "user" ? "justify-end" : "justify-start"}`}
   577→          >
   578→            <div
   579→              className={`max-w-[85%] whitespace-pre-wrap rounded-2xl px-3.5 py-2.5 text-sm leading-relaxed ${
   580→                m.role === "user"
   581→                  ? "bg-[#D4AF37] text-[#0F0F0F]"
   582→                  : "border border-white/8 bg-[#0F0F0F] text-white/90"
   583→              }`}
   584→            >
   585→              {m.content}
   586→            </div>
   587→          </div>
   588→        ))}
   589→        {loading && (
   590→          <div className="flex justify-start">
   591→            <div className="inline-flex items-center gap-2 rounded-2xl border border-white/8 bg-[#0F0F0F] px-3.5 py-2.5 text-sm text-[#A1A1A1]">
   592→              <Loader2 className="size-3.5 animate-spin text-[#D4AF37]" />
   593→              Sto pensando…
   594→            </div>
   595→          </div>
   596→        )}
   597→      </div>
   598→      <div className="flex gap-2">
   599→        <input
   600→          type="text"
   601→          value={input}
   602→          onChange={(e) => setInput(e.target.value)}
   603→          onKeyDown={(e) => {
   604→            if (e.key === "Enter") {
   605→              e.preventDefault();
   606→              send();
   607→            }
   608→          }}
   609→          placeholder="Scrivi una domanda…"
   610→          className="h-11 flex-1 rounded-full border border-white/12 bg-[#1C1C1C] px-4 text-sm text-white placeholder:text-[#6B6B6B] focus:border-[#D4AF37] focus:outline-none"
   611→        />
   612→        <button
   613→          type="button"
   614→          onClick={send}
   615→          disabled={!input.trim() || loading}
   616→          className="inline-flex h-11 items-center justify-center rounded-full bg-[#D4AF37] px-5 text-xs font-semibold uppercase tracking-[0.16em] text-[#0F0F0F] disabled:opacity-40"
   617→        >
   618→          Invia
   619→        </button>
   620→      </div>
   621→    </div>
   622→  );
   623→}
